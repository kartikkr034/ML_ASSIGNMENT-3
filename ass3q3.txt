import pandas as pd
import numpy as np

from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, MinMaxScaler

# Create numerical dataset
data = {
    "Age": [22, 25, np.nan, 30, 28],
    "Salary": [25000, 32000, 28000, np.nan, 45000],
    "Experience": [1, 2, 3, 5, 7]
}

df = pd.DataFrame(data)

print("--- Original Data ---")
print(df)

# Fill missing values using median
imputer = SimpleImputer(strategy="median")

X_imputed = imputer.fit_transform(df)

print("\n--- Data After Imputation ---")
print(X_imputed)

# StandardScaler
standard_scaler = StandardScaler()
X_standard = standard_scaler.fit_transform(X_imputed)

print("\n--- StandardScaler Result ---")
print(X_standard)

print("\nStandardScaler Minimum Values:")
print(X_standard.min(axis=0))

print("StandardScaler Maximum Values:")
print(X_standard.max(axis=0))

# MinMaxScaler
minmax_scaler = MinMaxScaler()
X_minmax = minmax_scaler.fit_transform(X_imputed)

print("\n--- MinMaxScaler Result ---")
print(X_minmax)

print("\nMinMaxScaler Minimum Values:")
print(X_minmax.min(axis=0))

print("MinMaxScaler Maximum Values:")
print(X_minmax.max(axis=0))
